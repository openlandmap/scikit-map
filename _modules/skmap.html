from abc import ABC, abstractmethod

from skmap.misc import ttprint

class SKMapBase(ABC):
    """Abstract base class for all skmap runners, providing a ``verbose`` helper."""

    def _verbose(self, *args, **kwargs) -> None:
        if self.verbose:
            ttprint(*args, **kwargs)
